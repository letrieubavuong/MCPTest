\begin{ex}
Tính đạo hàm của hàm số $f(x)=x^2$.
\choice{$x$}{\True $2x$}{$x^3$}{$2$}
\loigiai{Ta có $f'(x)=2x$.}
\end{ex}

\end{ex}

\begin{ex}
Tính $\int_0^1 2x\,dx$.
\shortans{1}
\loigiai{$\int_0^1 2x\,dx=[x^2]_0^1=1$.}
\end{ex}
